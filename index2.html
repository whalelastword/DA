# Статистический тест связи между полом и совершением покупок
table = pd.crosstab(df['SEX'], df['pk2'])
chi2, p, dof, expected = chi2_contingency(table)
print(f"\nРезультаты χ²-теста (пол и онлайн-покупки):")
print(f"  χ² = {chi2:.4f}")
print(f"  p-value = {p:.4f}")
if p < 0.05:
    print("  Вывод: Обнаружена статистически значимая связь между полом и онлайн-покупками (p < 0.05)")
else:
    print("  Вывод: Статистически значимой связи между полом и онлайн-покупками не обнаружено (p ≥ 0.05)")
